-- 1. LEADS
create table if not exists leads (
  id                  uuid primary key default gen_random_uuid(),
  created_at          timestamptz default now(),
  name                text not null,
  title               text,
  company             text,
  industry            text,
  location            text,
  email               text,
  email_status        text default 'unknown',
  linkedin_url        text,
  icp_score           integer default 0,
  status              text default 'new',
  kanban_column       text default 'New',
  source              text default 'linkedin',
  notes               text,
  sequence_id         uuid,
  client_id           uuid,
  last_touched        timestamptz default now(),
  scoring_reasoning   jsonb,
  recommended_action  text
);

-- email_status  : 'verified' | 'risky' | 'unknown'
-- status        : 'new' | 'contacted' | 'replied' | 'hot' | 'meeting' | 'won' | 'lost'
-- source        : 'linkedin' | 'google_maps' | 'amazon' | 'manual'
-- kanban_column : 'New' | 'Contacted' | 'Replied' | 'Hot Lead' | 'Meeting Booked' | 'Closed Won' | 'Closed Lost'

-- 2. MESSAGES
create table if not exists messages (
  id            uuid primary key default gen_random_uuid(),
  created_at    timestamptz default now(),
  lead_id       uuid references leads(id) on delete cascade,
  message_type  text,
  tone          text,
  subject       text,
  body          text not null,
  char_count    integer
);
-- message_type: 'linkedin_connection' | 'linkedin_dm' | 'cold_email' | 'followup_email' | 'breakup_email'

-- 3. SEQUENCES
create table if not exists sequences (
  id                  uuid primary key default gen_random_uuid(),
  created_at          timestamptz default now(),
  name                text not null,
  steps               jsonb not null default '[]',
  assigned_lead_ids   jsonb default '[]'
);
-- steps item: { day: number, channel: 'linkedin'|'email', type: string, template: string, active: boolean }

-- 4. CAMPAIGNS
create table if not exists campaigns (
  id                uuid primary key default gen_random_uuid(),
  created_at        timestamptz default now(),
  name              text not null,
  target_industry   text,
  status            text default 'active',
  lead_ids          jsonb default '[]'
);
-- status: 'active' | 'paused' | 'complete'

-- 5. CLIENTS
create table if not exists clients (
  id                  uuid primary key default gen_random_uuid(),
  created_at          timestamptz default now(),
  name                text not null,
  company             text,
  industry            text,
  monthly_retainer    integer default 0,
  status              text default 'active'
);

-- 6. ACTIVITY LOG
create table if not exists activity_log (
  id          uuid primary key default gen_random_uuid(),
  created_at  timestamptz default now(),
  type        text,
  text        text not null,
  lead_id     uuid
);
-- type: 'lead_added' | 'message_sent' | 'scored_hot' | 'meeting_booked' | 'lead_moved' | 'notification'

API CALL IMPLEMENTATION:

  const res = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'x-api-key': apiKey,
      'anthropic-version': '2023-06-01',
      'anthropic-dangerous-direct-browser-access': 'true'
    },
    body: JSON.stringify({
      model: 'claude-sonnet-4-20250514',
      max_tokens: 1000,
      system: `You are an expert B2B sales copywriter for LinkedIn + Email outreach
               specializing in SaaS, Founders, and Agencies.
               Never use "I hope this finds you well" or similar clichés.
               Always reference something specific about the person or company.
               Be direct, human, value-first. No buzzwords.`,
      messages: [{ role: 'user', content: buildPrompt(lead, type, tone, offer) }]
    })
  })

  function buildPrompt(lead, messageType, tone, offer) {
    const base = `Write a ${messageType} with ${tone} tone for:
      Name: ${lead.name} | Title: ${lead.title}
      Company: ${lead.company} | Industry: ${lead.industry}
      Offer: ${offer}
      Notes: ${lead.notes || 'none'}`

    if (messageType === 'linkedin_connection')
      return base + '\n\nMax 300 characters. No hashtags. Reference something specific.'
    if (messageType === 'cold_email')
      return base + '\n\nReturn ONLY valid JSON: {"subject":"...","body":"..."}'
    return base
  }

- All transitions: 150–200ms ease on interactive elements
- Border radius: 6–8px cards, 8px inputs, 12px modals
